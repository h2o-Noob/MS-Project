% ---------- 2. problem ----------
\begin{frame}{Why model LLM serving as a queueing network?}
  \begin{itemize}
    \item Requests wait at admission and tokenization.
    \item Requests wait for batching.
    \item Requests wait for GPU service.
  \end{itemize}
  \vfill
  \centering
  {\normalsize\textcolor{MutedColor}{the scheduler knobs}}\quad{\Large\textcolor{PrColor}{$(B,\ \tau,\ c)$}}
  \vfill
  \raggedright
  {\color{PrColor}\bfseries\itshape These knobs change where requests wait.}
  \smallskip

  A validated simulator lets us study these trade-offs reproducibly.
\end{frame}

% ---------- 3. model ----------
\begin{frame}{The three-stage queueing model}
  \begin{columns}[T]
    \begin{column}{195pt}
      {\color{PrColor}\bfseries Q1, admission and tokenization.}\\
      Geo/D/1, one server, deterministic service.
      \smallskip

      {\color{IndColor}\bfseries Q2, batching scheduler.}\\
      FCFS; a batch is released when $B$ requests have collected, or when the oldest has waited $\tau$ cycles. $B = 1$: pass-through.
      \smallskip

      {\color{SecColor}\bfseries Q3, GPU workers.}\\
      $c$ parallel workers; one worker serves one batch, all its requests share the duration.
    \end{column}
    \begin{column}{172pt}
      \centering
      {\large $S_1 = 2$ cycles}
      \medskip

      {\large\[
        S_{\mathrm{batch}}(b) = \left\lceil S_3 + \delta\,(b-1) \right\rceil
      \]}
      {\large\[
        \rho_3 = \frac{\lambda_{\mathrm{batch}}\, E[S_{\mathrm{batch}}]}{c} < 1
      \]}
      \smallskip
      {\small $B = 1$: \ $\rho_3 = \lambda_3\, E[S_3] / c < 1$}
      \smallskip
      {\scriptsize\textcolor{MutedColor}{$\lambda_{\mathrm{batch}}$: Q2 batch-dispatch rate. Single-server nodes: $\rho_i = \lambda_i E[S_i] < 1$.}}
    \end{column}
  \end{columns}
  \vfill
  \centering\small Bernoulli($\lambda$) arrivals; one cycle = one scheduler tick.
\end{frame}

% ---------- 4. simulator ----------
\begin{frame}{Simulator and cycle semantics}
  \begin{enumerate}
    \item \textbf{\textcolor{PrColor}{Arrival}} \; Bernoulli($\lambda$) request joins the queue
    \item \textbf{\textcolor{PrColor}{Service countdown}} \; busy workers count down or complete
    \item \textbf{\textcolor{PrColor}{Completion and routing}} \; finished work moves downstream
    \item \textbf{\textcolor{PrColor}{Dispatch}} \; idle servers take the head of the line
    \item \textbf{\textcolor{PrColor}{Counters}} \; occupancy, delay, utilization update
  \end{enumerate}
  \vfill
  One scheduler tick = one simulation cycle: fixed-increment time advance, the course's cycle-driven scheme. Same-cycle immediate access: a request arriving at cycle $t$ can enter service at cycle $t$.
\end{frame}

% ---------- 5. analytical validation ----------
\begin{frame}{Validation against analytical queueing models}
  \centering
  {\large\[
    \rho = \lambda S
    \qquad\qquad
    L_q = \lambda W_q
  \]}
  {\large\[
    W_q = \frac{\rho\,(S-1)}{2\,(1-\rho)}
    \qquad\qquad
    L_q^{\mathrm{Geo/Geo/1}} = \frac{2\lambda^2}{1 - 2\lambda}
  \]}
  \vfill
  \raggedright
  Anchors: single-station configurations of the engine are exactly our validated single-queue simulators.
  \smallskip

  {\small The $W_q$ expression is the general one for our slotted convention; extending the sweep to $S = 3$ showed that the $L_q$ form we had initially carried over is {\color{ObsColor}specific to $S = 2$}.}
  \smallskip

  {\small\textcolor{MutedColor}{Criterion: worst relative error below $1\%$ at $10^7$ cycles: $\lambda \le 0.45$ for $S = 2$, $\lambda \le 0.30$ for $S = 3$; lengths $10^3$, $10^5$, $10^7$.}}
\end{frame}

  \medskip

  In the tested grid, $B = 1$ has the lowest end-to-end delay.

% ---------- 11. key takeaways ----------
\colorlet{RuleAccent}{InkColor}
\begin{frame}{Key takeaways}
  \begin{enumerate}
    \item \textbf{\textcolor{PrColor}{Validated simulator.}}\ Single-queue analytical checks within $1\%$ at converged loads; network accounting to $3.4\times 10^{-7}$.
    \item \textbf{\textcolor{SecColor}{Variability and parallelism shape latency.}}\ Geometric GPU service increases delay strongly at one worker; a second worker largely absorbs it.
    \item \textbf{\textcolor{IndColor}{Batching trades delay for efficiency.}}\ Larger batches improve utilization but increase waiting; saturation near $B = 8$ for $\tau = 8$; $B = 1$ delays least in the tested grid.
  \end{enumerate}
  \vfill
  {\scriptsize\textcolor{MutedColor}{Limitations: linear feedforward chain; single Bernoulli input; static batching; batch-level service abstraction; two service laws; tick-quantized time; cycle mapping not calibrated. Network-level analytical validation remains open pending course material.}}  \vfill
  \centering\textbf{Thank you.}
\end{frame}
